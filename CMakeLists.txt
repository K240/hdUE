cmake_minimum_required(VERSION 3.24)
project(hdUE VERSION 0.1.0 LANGUAGES C CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "hdUE currently requires Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")

add_library(hdue_client STATIC src/client/backend.cpp src/client/image.cpp src/client/metrics.cpp)
target_link_libraries(hdue_client PUBLIC dxgi)
target_include_directories(hdue_client PUBLIC include)
target_compile_definitions(hdue_client PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(hdue_client PRIVATE /W4 /permissive- /utf-8)

add_executable(hdue_backend_smoke src/host/backend_smoke.cpp)
target_link_libraries(hdue_backend_smoke PRIVATE hdue_client user32)
target_compile_options(hdue_backend_smoke PRIVATE /W4 /permissive- /utf-8)
target_link_options(hdue_backend_smoke PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/host/hdue.manifest")

include(CTest)
set(HDUE_HOUDINI_ROOT "" CACHE PATH "Build against the Houdini 22.0.466 HDK instead of standalone OpenUSD")
if(HDUE_HOUDINI_ROOT)
    include(cmake/Houdini.cmake)
    return()
endif()
option(HDUE_BUILD_HYDRA "Build the OpenUSD 26.03 render delegate" ON)
set(OPENUSD_ROOT "$ENV{HDUE_OPENUSD_ROOT}" CACHE PATH "OpenUSD 26.03 prefix")
if(HDUE_BUILD_HYDRA)
    if(NOT OPENUSD_ROOT)
        message(FATAL_ERROR "Set OPENUSD_ROOT or HDUE_OPENUSD_ROOT to your OpenUSD 26.03 installation.")
    endif()
    list(PREPEND CMAKE_PREFIX_PATH "${OPENUSD_ROOT}")
    find_package(OpenGL REQUIRED)
    find_package(pxr CONFIG REQUIRED PATHS "${OPENUSD_ROOT}" NO_DEFAULT_PATH)
    if(NOT PXR_VERSION STREQUAL "2603")
        message(FATAL_ERROR "hdUE is pinned to OpenUSD 26.03; selected ${PXR_VERSION}.")
    endif()
    add_library(hdUE SHARED src/hydra/delegate.cpp src/hydra/prims.cpp src/hydra/render_buffer.cpp src/hydra/plugin.cpp src/hydra/picking.cpp)
    target_compile_definitions(hdUE PRIVATE HDUE_HYDRA_EXPORTS)
    target_compile_definitions(hdUE PRIVATE _SILENCE_CXX20_OLD_SHARED_PTR_ATOMIC_SUPPORT_DEPRECATION_WARNING)
    target_include_directories(hdUE PUBLIC src/hydra "${OPENUSD_ROOT}/include")
    target_link_libraries(hdUE PRIVATE hdue_client hd hf glf usdImaging usdGeom usdShade usdLux usd tf plug vt gf)
    target_compile_options(hdUE PRIVATE /W3 /utf-8 /bigobj)
    set(HDUE_PLUGIN_LIBRARY "Release/hdUE.dll")
    set(HDUE_PLUGIN_NAME hdUE)
    set(HDUE_PLUGIN_TYPE HdUERendererPlugin)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/resources")
    configure_file(resources/plugInfo.json.in "${CMAKE_BINARY_DIR}/resources/plugInfo.json" @ONLY)
    add_executable(hdue_usd_render src/host/usd_render.cpp)
    target_include_directories(hdue_usd_render PRIVATE "${OPENUSD_ROOT}/include")
    target_link_libraries(hdue_usd_render PRIVATE hdue_client hd hf usdImaging usdGeom usdShade usdLux usd tf plug vt gf user32)
    target_compile_options(hdue_usd_render PRIVATE /W3 /utf-8 /bigobj)
    target_compile_definitions(hdue_usd_render PRIVATE _SILENCE_CXX20_OLD_SHARED_PTR_ATOMIC_SUPPORT_DEPRECATION_WARNING)
    target_link_options(hdue_usd_render PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/host/hdue.manifest")
    if(BUILD_TESTING)
        add_executable(hdue_buffer_test tests/render_buffer.cpp src/hydra/render_buffer.cpp)
        target_include_directories(hdue_buffer_test PRIVATE src/hydra include "${OPENUSD_ROOT}/include")
        target_compile_definitions(hdue_buffer_test PRIVATE HDUE_HYDRA_EXPORTS _SILENCE_CXX20_OLD_SHARED_PTR_ATOMIC_SUPPORT_DEPRECATION_WARNING)
        target_link_libraries(hdue_buffer_test PRIVATE hd)
        add_test(NAME render_buffer_snapshots COMMAND hdue_buffer_test)
        add_executable(hdue_pick_test tests/picking.cpp src/hydra/picking.cpp)
        target_include_directories(hdue_pick_test PRIVATE src/hydra "${OPENUSD_ROOT}/include")
        target_link_libraries(hdue_pick_test PRIVATE gf)
        add_test(NAME picking_bvh COMMAND hdue_pick_test)
        get_filename_component(_python_lib_dir "${Python3_LIBRARY}" DIRECTORY)
        get_filename_component(_python_runtime "${_python_lib_dir}" DIRECTORY)
        set_tests_properties(render_buffer_snapshots PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:${OPENUSD_ROOT}/lib;PATH=path_list_prepend:${OPENUSD_ROOT}/bin;PATH=path_list_prepend:${_python_runtime}")
        set_tests_properties(picking_bvh PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:${OPENUSD_ROOT}/lib;PATH=path_list_prepend:${OPENUSD_ROOT}/bin;PATH=path_list_prepend:${_python_runtime}")
    endif()
endif()
if(BUILD_TESTING)
    add_executable(hdue_abi_layout tests/abi_layout.c)
    target_include_directories(hdue_abi_layout PRIVATE include)
    add_test(NAME abi_c_layout COMMAND hdue_abi_layout)
endif()
